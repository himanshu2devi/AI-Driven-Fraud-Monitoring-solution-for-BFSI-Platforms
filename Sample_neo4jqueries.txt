3)THIS is exactly your graph risk source.


MATCH (a:Account)-[r:TRANSFERRED]->()

WHERE r.status = 'FRAUD'

WITH a, COUNT(r) AS fraudCount

RETURN
a.accountNumber AS Account,

fraudCount,

CASE
WHEN fraudCount >= 5 THEN 'HIGH'
WHEN fraudCount >= 3 THEN 'MEDIUM'
ELSE 'LOW'
END AS RiskLevel

ORDER BY fraudCount DESC

:Neo4j graph intelligence dynamically
calculates account risk level based
on historical fraud relationships.

MATCH (a)-[fr:TRANSFERRED]->()

WHERE fr.status = 'FRAUD'

WITH
a,
b,
r,
COUNT(fr) AS fraudCount

RETURN

a.accountNumber AS FromAccount,

b.accountNumber AS ToAccount,

r.amount AS Amount,

r.status AS Status,

r.reason AS Reason,

r.timestamp AS Timestamp,

fraudCount,

CASE
WHEN fraudCount >= 10 THEN 0.9
WHEN fraudCount >= 5 THEN 0.6
WHEN fraudCount >= 2 THEN 0.3
ELSE 0.0
END AS GraphRiskScore



7)latest 10 risky accounts:
MATCH (a:Account)-[r:TRANSFERRED]->()

WHERE r.status = 'FRAUD'

WITH
a,
COUNT(r) AS fraudCount,
MAX(r.timestamp) AS latestTxn

RETURN

a.accountNumber AS Account,

fraudCount,

latestTxn,

CASE
WHEN fraudCount >= 10 THEN 'HIGH'
WHEN fraudCount >= 5 THEN 'MEDIUM'
WHEN fraudCount >= 2 THEN 'LOW'
ELSE 'MINIMAL'
END AS RiskLevel

ORDER BY latestTxn DESC
LIMIT 5
